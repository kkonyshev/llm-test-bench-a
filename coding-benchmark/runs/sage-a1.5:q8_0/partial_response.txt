```python
"""Task Tracker API — FastAPI + SQLite."""

from contextlib import asynccontextmanager
from typing import Optional

import sqlite3
from datetime import datetime

from fastapi import Depends, FastAPI, HTTPException, Query
from pydantic import BaseModel, field_validator


VALID_PRIORITIES = ("low", "medium", "high")
DB_FILE = "tasks.db"

CREATE_TABLE_SQL = """
CREATE TABLE IF NOT EXISTS tasks (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    title TEXT NOT NULL,
    description TEXT,
    completed INTEGER NOT NULL DEFAULT 0,
    priority TEXT NOT NULL DEFAULT 'medium',
    created_at TEXT NOT NULL
)"""


def create_tables():
    conn = sqlite3.connect(DB_FILE)
    try:
        conn.execute(CREATE_TABLE_SQL)
        conn.commit()
    finally:
        conn.close()


def get_db():
    conn = sqlite3.connect(DB_FILE)
    conn.row_factory = sqlite3.Row
    try:
        yield conn
    finally:
        conn.close()


def row_to_dict(row):
    return {key: row[key] for key in row.keys()}


class TaskCreate(BaseModel):
    title: str
    description: Optional[str] = None
    priority: str = "medium"

    @field_validator("title")
    def validate_title(cls, value):
        value = value.strip()
        if not value or len(value) > 200:
            raise ValueError("title must be 1-200 non-empty characters")
        return value

    @field_validator("priority")
    def validate_priority(cls, value):
        if value not in VALID_PRIORITIES:
            raise ValueError(f"priority must be one of {', '.join(VALID_PRIORITIES)}")
        return value


class TaskUpdate(BaseModel):
    title: Optional[str] = None
    description: Optional[str] = None
    completed: Optional[bool] = None
    priority: Optional[str] = None

    @field_validator("title")
    def validate_title(cls, value):
        if value is None:

        value = value.strip()
        if not value or len(value) > 200:

        return value

    @field_validator("priority")
    def validate_priority(cls, value):
        if value is not None and value not in VALID_PRIORITIES:
            raise ValueError(f"priority must be one of {', '.join(VALID_PRIORITIES)}")
        return value


class TaskResponse(BaseModel):
    id: int
    title: str
    description: Optional[str] = None
    completed: bool
    priority: str
    created_at: str
```

```python
    description: Optional[str] = None
```
```python
    priority: str = "medium"

    @field_validator("title")
    def validate_title(cls, value):
        value = value.strip()
        if not value or len(value) > 200:
            raise ValueError("title must be a non-empty string of at most 200 characters")
        return value

    @field_validator("priority")
    def validate_priority(cls, value):
        if value not in VALID_PRIORITIES:
            raise ValueError(f"priority must be one of {', '.join(VALID_PRIORITIES)}")
        return value


class TaskUpdate(BaseModel):
    title: Optional[str] = None
    description: Optional[str] = None
    completed: Optional[bool] = None
    priority: Optional[str] = None

    @field_validator("title")
    def validate_title(cls, value):
        if value is None:
            return value
        value = value.strip()
        if not value or len(value) > 200:
            raise ValueError("title must be a non-empty string of at most 200 characters")
        return value

    @field_validator("priority")
    def validate_priority(cls, value):
        if value is not None and value not in VALID_PRIORITIES:
            raise ValueError(f"priority must be one of {', '.join(VALID_PRIORITIES)}")
        return value


class TaskResponse(BaseModel):
    id: int
    title: str
    description: Optional[str] = None
    completed: bool
    priority: str
    created_at: str


@asynccontextmanager
async def lifespan(app: FastAPI):
    create_tables()
    yield


app = FastAPI(lifespan=lifespan)


@app.post("/tasks", status_code=201, response_model=TaskResponse)
async def create_task(task: TaskCreate, db: sqlite3.Connection = Depends(get_db)):
    now = datetime.now().isoformat()
    db.execute(
        "INSERT INTO tasks (title, description, completed, priority, created_at) VALUES (?, ?, ?, ?, ?)",
        (task.title, task.description, False, task.priority, now),
    )
    db.commit()
    row = db.execute("SELECT * FROM tasks WHERE id = ?", (db.lastrowid,)).fetchone()
    return TaskResponse(**row_to_dict(row))
```
