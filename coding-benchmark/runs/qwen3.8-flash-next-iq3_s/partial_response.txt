async def read_json_object(request: Request, allow_empty: bool = False) -> dict:
    try:

    except Exception:

    if not isinstance(data, dict):

def init_db() -> None:

app = FastAPI()

@app.on_event("startup")

    init_db()

init_db()

async def read_json_object(request: Request, allow_empty: bool = False) -> dict:
    try:
        raw_body = await request.body()
    except Exception:
        raise HTTPException(status_code=422, detail="Invalid request body")
    if allow_empty and raw_body.strip() == b"":
        return {}
    try:
        data = json.loads(raw_body)
    except Exception:
        raise HTTPException(status_code=422, detail="Invalid JSON body")
    if not isinstance(data, dict):
        raise HTTPException(status_code=422, detail="Request body must be a JSON object")
    return data

def validate_title(value: object) -> str:
    if not isinstance(value, str):
        raise HTTPException(status_code=422, detail="title must be a string")
    stripped = value.strip()
    if not stripped or len(stripped) > 200:
        raise HTTPException(status_code=422, detail="title must be between 1 and 200 characters after trimming")
    return stripped

def validate_description(value: object) -> Optional[str]:
    if value is not None and not isinstance(value, str):
        raise HTTPException(status_code=422, detail="description must be a string or null")
    return value

def validate_priority(value: object) -> str:
    if not isinstance(value, str) or value not in ALLOWED_PRIORITIES:
        raise HTTPException(status_code=422, detail="priority must be one of low, medium, high")
    return value

def parse_bool_query(value: Optional[str]) -> Optional[bool]:
    if value is None:
        return None
    lowered = value.lower()
    if lowered in ("true", "1"):
        return True
    if lowered in ("false", "0"):
        return False
    raise HTTPException(status_code=422, detail="completed must be a boolean")

def task_from_row(row: sqlite3.Row) -> dict:
    return {
        "id": row["id"],
        "title": row["title"],
        "description": row["description"],
        "completed": bool(row["completed"]),
        "priority": row["priority"],
        "created_at": row["created_at"],
    }

@app.post("/tasks", status_code=201)
async def create_task(request: Request) -> dict:
    payload = await read_json_object(request, allow_empty=True)
    if "title" not in payload:
        raise HTTPException(status_code=422, detail="title is required")
    title = validate_title(payload["title"])
    description = validate_description(payload.get("description")) if "description" in payload else None
    priority = validate_priority(payload["priority"]) if "priority" in payload else "medium"
    created_at = datetime.now(timezone.utc).isoformat()
    conn = get_connection()
    try:
        cursor = conn.execute(
            "INSERT INTO tasks (title, description, completed, priority, created_at) VALUES (?, ?, ?, ?, ?)",
            (title, description, 0, priority, created_at),
        )
        conn.commit()
        row = conn.execute("SELECT * FROM tasks WHERE id = ?", (cursor.lastrowid,)).fetchone()
    finally:
        conn.close()
    return task_from_row(row)

import json
import sqlite3
from datetime import datetime, timezone
from typing import Optional

from fastapi import FastAPI, HTTPException, Query, Request, Response

DB_PATH = "tasks.db"
ALLOWED_PRIORITIES = ("low", "medium", "high")

CREATE_TABLE_SQL = """
CREATE TABLE IF NOT EXISTS tasks (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    title TEXT NOT NULL,
    description TEXT,
    completed INTEGER NOT NULL DEFAULT 0,
    priority TEXT NOT NULL DEFAULT 'medium',
    created_at TEXT NOT NULL
)
"""


def get_connection() -> sqlite3.Connection:
    conn = sqlite3.connect(DB_PATH, timeout=30.0)
    conn.row_factory = sqlite3.Row
    return conn


def init_db() -> None:
    conn = get_connection()
    try:
        conn.execute(CREATE_TABLE_SQL)
        conn.commit()
    finally:
        conn.close()


app = FastAPI()


@app.on_event("startup")
def startup_event() -> None:
    init_db()


init_db()


async def read_json_object(request: Request, allow_empty: bool = False) -> dict:
    try:
        body = await request.body()
    except Exception:
        raise HTTPException(status_code=422, detail="Invalid request body")

    if allow_empty and body.strip() == b"":
        return {}

    try:
        data = json.loads(body)
    except Exception:
        raise HTTPException(status_code=422, detail="Invalid JSON body")

    if not isinstance(data, dict):
        raise HTTPException(status_code=422, detail="Request body must be a JSON object")

    return data


def validate_title(value: object) -> str:
    if not isinstance(value, str):
        raise HTTPException(status_code=422, detail="title must be a string")

    stripped = value.strip()
    if not stripped or len(stripped) > 200:
        raise HTTPException(
            status_code=422,
            detail="title must be between 1 and 200 characters after trimming",
        )

    return stripped


def validate_description(value: object) -> Optional[str]:
    if value is not None and not isinstance(value, str):
        raise HTTPException(status_code=422, detail="description must be a string or null")
    return value


def validate_priority(value: object) -> str:
    if not isinstance(value, str) or value not in ALLOWED_PRIORITIES:
        raise HTTPException(status_code=422, detail="priority must be one of low, medium, high")
    return value


def parse_bool_query(value: Optional[str]) -> Optional[bool]:
    if value is None:
        return None

    value = value.strip().lower()
    if value in ("true", "1"):
        return True
    if value in ("false", "0"):
        return False

    raise HTTPException(status_code=422, detail="completed must be a boolean")


def task_from_row(row: sqlite3.Row) -> dict:
    return {
        "id": row["id"],
        "title": row["title"],
        "description": row["description"],
        "completed": bool(row["completed"]),
        "priority": row["priority"],
        "created_at": row["created_at"],
    }


@app.post("/tasks", status_code=201)
async def create_task(request: Request):
    payload = await read_json_object(request, allow_empty=True)

    if "title" not in payload:
        raise HTTPException(status_code=422, detail="title is required")

    title = validate_title(payload["title"])
    description = (
        validate_description(payload["description"]) if "description" in payload else None
    )
    priority = (
        validate_priority(payload["priority"]) if "priority" in payload else "medium"
    )

    created_at = datetime.now(timezone.utc).isoformat()

    conn = get_connection()
    try:
        cursor = conn.execute(
            """
            INSERT INTO tasks (title, description, completed, priority, created_at)
            VALUES (?, ?, ?, ?, ?)
            """,
            (title, description, 0, priority, created_at),
        )
        conn.commit()
        row = conn.execute("SELECT * FROM tasks WHERE id = ?", (cursor.lastrowid,)).fetchone()
    finally:
        conn.close()

    return task_from_row(row)
