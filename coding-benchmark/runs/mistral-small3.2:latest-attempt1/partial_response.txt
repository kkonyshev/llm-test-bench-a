from fastapi import FastAPI, HTTPException, status
from pydantic import BaseModel, Field, constr
from typing import Optional, List
from datetime import datetime
import sqlite3
from sqlite3 import Connection
from contextlib import contextmanager

app = FastAPI()

DATABASE_URL = "tasks.db"
PRIORITY_OPTIONS = ["low", "medium", "high"]

class Task(BaseModel):
    id: int
    title: str = Field(..., min_length=1, max_length=200)
    description: Optional[str] = None
    completed: bool = False
    priority: str = Field("medium", constr(to_lower=True))
    created_at: datetime = Field(default_factory=datetime.now)

class TaskCreate(BaseModel):
    title: str = Field(..., min_length=1, max_length=200)
    description: Optional[str] = None
    priority: Optional[str] = Field(None, constr(to_lower=True))

class TaskUpdate(BaseModel):
    title: Optional[str] = Field(None, min_length=1, max_length=200)
    description: Optional[str] = None
    completed: Optional[bool] = None
    priority: Optional[str] = Field(None, constr(to_lower=True))

@contextmanager
def get_db():
    conn = sqlite3.connect(DATABASE_URL)
    try:
        yield conn
    finally:
        conn.close()

def init_db():
    with get_db() as conn:
        cursor = conn.cursor()
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS tasks (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                title TEXT NOT NULL,
                description TEXT,
                completed BOOLEAN DEFAULT FALSE,
                priority TEXT DEFAULT 'medium',
                created_at TEXT NOT NULL
            )
        """)
        conn.commit()

init_db()

@app.post("/tasks", status_code=status.HTTP_201_CREATED)
async def create_task(task: TaskCreate):
    if task.priority and task.priority not in PRIORITY_OPTIONS:
        raise HTTPException(
            status_code=status.HTTP_422_UNPROCESSABLE_ENTITY,
            detail="Priority must be one of: low, medium, high"
        )

    with get_db() as conn:
        cursor = conn.cursor()
        cursor.execute("""
            INSERT INTO tasks (title, description, priority, created_at)
            VALUES (?, ?, ?, ?)
        """, (
            task.title.strip(),
            task.description,
            task.priority if task.priority else "medium",
            datetime.now().isoformat()
        ))
        conn.commit()
        task_id = cursor.lastrowid

    return Task(
        id=task_id,
        title=task.title.strip(),
        description=task.description,
        completed=False,
        priority=task.priority if task.priority else "medium",
        created_at=datetime.now()
    )
